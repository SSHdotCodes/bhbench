cmake_minimum_required(VERSION 3.16)
project(BlackHole LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
find_package(OpenGL REQUIRED)
find_package(glfw3 3.3 REQUIRED)
add_executable(black-hole src/main.mm)
target_compile_definitions(black-hole PRIVATE GL_SILENCE_DEPRECATION SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
target_link_libraries(black-hole PRIVATE glfw OpenGL::GL "-framework Cocoa" "-framework CoreText" "-framework CoreGraphics")
add_executable(physics-tests tests/physics.cpp)
enable_testing()
add_test(NAME physics COMMAND physics-tests)
